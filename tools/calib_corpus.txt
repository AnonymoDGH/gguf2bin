typedef struct node { int value; struct node *next; } node_t;

static node_t *push(node_t *head, int value) {
    node_t *n = malloc(sizeof *n);
    if (!n) return head;
    n->value = value;
    n->next = head;
    return n;
}

    return 0;
}

def word_count(path):
    """Return the ten most common words in a text file."""
    with open(path, encoding="utf-8") as f:
